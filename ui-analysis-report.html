_enterGame() 详细步骤

function _enterGame() {
  // 1. 停止登录页 demo 动画
  SnakeGame.stopDemo();

  // 2. DOM 视图切换
  authOverlay.style.display = 'none';
  navbar.classList.add('visible');
  showView('game');

  // 3. 更新导航栏用户名
  navUsername.textContent = `👤 ${user}`;

  // 4. 初始化游戏引擎（挂载 Canvas + 注册回调）
  SnakeGame.init('game-canvas', {
    onScoreUpdate: (score) => { scoreValue.textContent = score; },
    onGameOver: (score, duration) => {
      btnStart.textContent = '🔄 再来一局';
      DataStore.saveGameRecord(user, score, duration);
    }
  });
}

showView(name) — 视图路由函数

function showView(name) {
  // 隐藏所有视图
  [viewGame, viewLeaderboard].forEach(v => v.style.display = 'none');

  if (name === 'game')        viewGame.style.display = 'block';
  if (name === 'leaderboard') {
    viewLeaderboard.style.display = 'block';
    _refreshLeaderboard();   // 切换时刷新数据
  }
}

回调模式 — 游戏引擎与 UI 解耦

// SnakeGame.init() 接受回调，实现游戏引擎与 DOM 解耦
SnakeGame.init('game-canvas', {
  // 每吃一个食物触发
  onScoreUpdate: (score) => {
    scoreValue.textContent = score;   // app.js 负责 DOM 操作
  },
  // 游戏结束时触发
  onGameOver: (score, duration) => {
    btnStart.textContent = '🔄 再来一局';
    DataStore.saveGameRecord(currentUser, score, duration);
    // 注意：DataStore 调用也在 SnakeGame._endGame() 内部也有一份
    // app.js 侧的调用用于保持 UI 可扩展性
  }
});

// SnakeGame 内部 _endGame() 通知 app.js
function _endGame() {
  clearInterval(interval);
  gameOver = true;
  running = false;
  const duration = Math.floor((Date.now() - startTime) / 1000);
  callbacks.onGameOver(score, duration);  // 上报给 app.js
}
